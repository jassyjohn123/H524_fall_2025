\section*{R Code Reference}

\textbf{Chi-square goodness-of-fit:}
\begin{verbatim}
observed <- c(95, 28, 26, 11)
expected_props <- c(9/16, 3/16, 3/16, 1/16)
chisq.test(observed, p=expected_props)
\end{verbatim}

\textbf{Chi-square test of independence:}
\begin{verbatim}
data <- matrix(c(75, 425, 35, 465), nrow=2, byrow=TRUE)
chisq.test(data)
\end{verbatim}

\textbf{Fisher's exact test:}
\begin{verbatim}
data <- matrix(c(7, 3, 2, 8), nrow=2, byrow=TRUE)
fisher.test(data)
\end{verbatim}

\textbf{McNemar's test:}
\begin{verbatim}
data <- matrix(c(18, 12, 22, 8), nrow=2, byrow=TRUE)
mcnemar.test(data)
\end{verbatim}
